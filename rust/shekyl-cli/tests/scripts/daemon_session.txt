# Pattern for a daemon-backed CLI session. Open is `--wallet-file` on the
# invocation, so the password never sits in this file.
#
#   shekyl-cli --json --script-file tests/scripts/daemon_session.txt \
#     --network testnet --daemon-endpoint 127.0.0.1:<channel-port> --daemon-bundle-file <bundle> \
#     --wallet-file <dir>/<name>.wallet --wallet-password-file <path>
#
# The daemon is on the wallet's own network; a daemon on any other network,
# regtest included, is refused on identity.
#
# The test runs the script as written and checks INVALID_RECIPIENT. To pay
# from a funded wallet, replace <recipient> with a full testnet address.
balance
wallet refresh
chain
send 0.001 <recipient> --yes
